
def reflex_vacuum_agent(location, status):
    if status == 'Dirty':
        return 'Suck'
    elif location == 'A':
        return 'Right'
    elif location == 'B':
        return 'Left'


def simulate_environment(environment):
    location = 'A'

    print("Initial Environment:", environment)
    print("Initial Agent Location:", location)

    while 'Dirty' in environment.values():

        status = environment[location]

        action = reflex_vacuum_agent(location, status)

        print(f"\nLocation: {location}")
        print(f"Status: {status}")
        print(f"Action: {action}")

        if action == 'Suck':
            environment[location] = 'Clean'
            print(f"Location {location} cleaned.")

        elif action == 'Right':
            location = 'B'
            print("Agent moved to Location B.")

        elif action == 'Left':
            location = 'A'
            print("Agent moved to Location A.")

    print("\nFinal Environment:", environment)
    print("All locations are clean!")


# Example usage
env = {'A': 'Dirty', 'B': 'Dirty'}
simulate_environment(env)
