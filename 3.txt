
from collections import deque

# Generate all possible legal moves
def get_neighbors(state):
    neighbors = []

    for i in range(3):
        for j in range(3):

            # Check source peg is not empty
            if not state[i]:
                continue

            # Get top disk
            disk = state[i][-1]

            # Check destination peg
            if not state[j] or state[j][-1] > disk:

                new_state = [list(peg) for peg in state]

                # Move disk
                new_state[i].pop()
                new_state[j].append(disk)

                new_state = tuple(tuple(peg) for peg in new_state)

                move = (disk, chr(65 + i), chr(65 + j))

                neighbors.append((new_state, move))

    return neighbors


# Breadth First Search
def tower_of_hanoi_bfs(n):

    initial = (tuple(range(n, 0, -1)), (), ())
    goal = ((), (), tuple(range(n, 0, -1)))

    queue = deque([initial])

    visited = {initial}

    parent = {initial: None}
    moves = {}

    while queue:

        current = queue.popleft()

        # Goal test
        if current == goal:
            break

        # Generate successor states
        for next_state, move in get_neighbors(current):

            if next_state not in visited:

                visited.add(next_state)
                parent[next_state] = current
                moves[next_state] = move

                queue.append(next_state)

    # Reconstruct solution path
    path = []
    current = goal

    while parent[current] is not None:
        path.append((moves[current], current))
        current = parent[current]

    path.reverse()

    return initial, path, len(visited)


# Display state
def display_state(state):
    for name, peg in zip("ABC", state):
        print(name, ":", list(peg))


# Main program
n = int(input("Enter number of disks: "))

if n <= 0:
    print("Please enter a positive number.")

else:
    initial, path, explored = tower_of_hanoi_bfs(n)

    print("\nInitial State:")
    display_state(initial)

    print("\nSequence of Moves:")

    for step, (move, state) in enumerate(path, 1):

        disk, source, destination = move

        print(f"\nMove {step}: Disk {disk} from {source} to {destination}")

        display_state(state)

    print("\nGoal State Reached!")
    print("Total Moves:", len(path))
    print("States Explored:", explored)
